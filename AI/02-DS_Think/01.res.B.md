```javascript
const fs = require('fs').promises;

/**
 * 异步读取文件指定位置的字节
 * @param {string} filePath 文件路径
 * @param {number} position 读取位置
 * @returns {Promise<number|null>} 返回字节数值（0-255），超出文件范围返回null
 */
async function readByteAt(filePath, position) {
    const fd = await fs.open(filePath, 'r');
    try {
        const buffer = Buffer.alloc(1);
        const { bytesRead } = await fd.read(buffer, 0, 1, position);
        return bytesRead === 1 ? buffer[0] : null;
    } finally {
        await fd.close();
    }
}

class LazyDiffNode {
    /**
     * @param {{path: string, size: number}[]} filesWithSize 包含文件大小信息的文件列表
     * @param {number} currentPosition 当前比较位置
     */
    constructor(filesWithSize, currentPosition) {
        this.files = filesWithSize;
        this.currentPosition = currentPosition;
        this.children = new Map();
    }

    /**
     * 工厂方法，创建并验证文件大小
     * @param {string[]} filePaths 文件路径列表
     * @returns {Promise<LazyDiffNode>}
     */
    static async create(filePaths) {
        const filesWithSize = await Promise.all(
            filePaths.map(async path => ({
                path,
                size: (await fs.stat(path)).size
            }))
        );

        // 验证所有文件大小一致
        const uniqueSizes = new Set(filesWithSize.map(f => f.size));
        if (uniqueSizes.size > 1) {
            throw new Error('All files must have the same size');
        }

        return new LazyDiffNode(filesWithSize, 0);
    }

    /**
     * 启动异步比较过程
     * @returns {Promise<void>}
     */
    async startCompare() {
        if (this.files.length <= 1) return;

        const fileSize = this.files[0].size;
        let currentPos = this.currentPosition;
        let differingPos = null;
        let bytes = [];

        // 查找第一个差异位置
        while (differingPos === null && currentPos < fileSize) {
            bytes = await Promise.all(
                this.files.map(file => 
                    currentPos < file.size 
                        ? readByteAt(file.path, currentPos)
                        : null
                )
            );

            if (!this.allBytesEqual(bytes)) {
                differingPos = currentPos;
            } else {
                currentPos++;
            }
        }

        if (differingPos === null) return; // 所有文件完全相同

        // 更新节点状态并创建子节点
        this.currentPosition = differingPos;
        const groups = this.groupFilesByByte(bytes);

        for (const [byteValue, files] of groups) {
            const childNode = new LazyDiffNode(files, differingPos + 1);
            this.children.set(byteValue, childNode);
            await childNode.startCompare();
        }
    }

    /**
     * 判断所有字节是否相同
     * @param {(number|null)[]} bytes 
     */
    allBytesEqual(bytes) {
        const first = bytes[0];
        return bytes.every(b => b === first);
    }

    /**
     * 根据字节值对文件进行分组
     * @param {(number|null)[]} bytes 
     */
    groupFilesByByte(bytes) {
        return bytes.reduce((groups, byte, index) => {
            const key = byte !== null ? byte : 'EOF';
            if (!groups.has(key)) groups.set(key, []);
            groups.get(key).push(this.files[index]);
            return groups;
        }, new Map());
    }

    /**
     * 打印树结构（用于调试）
     */
    async printTree(prefix = '', isLast = true) {
        const current = `Pos ${this.currentPosition}: [${this.files.map(f => f.path.split('/').pop())}]`;
        console.log(prefix + (isLast ? '└─ ' : '├─ ') + current);
        
        const entries = Array.from(this.children.entries());
        for (let i = 0; i < entries.length; i++) {
            const [byte, child] = entries[i];
            const newPrefix = prefix + (isLast ? '   ' : '│  ');
            await child.printTree(
                newPrefix,
                i === entries.length - 1
            );
        }
    }
}

// 测试示例
async function main() {
    try {
        // 创建测试文件（实际使用时需要真实文件）
        const rootNode = await LazyDiffNode.create([
            'file1.bin',
            'file2.bin',
            'file3.bin'
        ]);
        
        await rootNode.startCompare();
        console.log('LazyDiffTree comparison completed:');
        await rootNode.printTree();
    } catch (err) {
        console.error('Error:', err);
    }
}

main();
```

### 实现说明：
1. **文件处理**：
   - 使用`fs.promises`进行异步文件操作
   - 每个节点存储文件路径和大小信息
   - 通过`readByteAt`实现精确的异步字节读取

2. **核心算法**：
   - 从当前位詈开始线性扫描
   - 使用`Promise.all`并行读取所有文件字节
   - 发现差异后立即进行分组并递归处理子节点

3. **树结构特性**：
   - 每个节点自动维护当前位置和子节点映射
   - 使用Map结构存储子节点，键为差异字节值
   - 包含调试用的树结构打印方法
