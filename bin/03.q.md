我们先来看一下一个简单的反相例子吧：

```javascript
class ByteMap {
	constructor() {
		// 使用 Buffer 作为快速索引表 (byte -> array index)
		this.B = Buffer.alloc(256);
		this.B.fill(255); // 初始化为 255 (表示空)
		
		// 使用 Array 存储实际的值，可以动态扩容
		this.A = [];
	}

	set(key, value) {
		if (typeof key !== 'number' || key < 0 || key > 255) {
			throw new Error('Key must be a byte (0-255)');
		}

		const index = this.B[key];
		if (index !== 255) {
			this.A[index] = value;
		} else {
			this.B[key] = this.A.length;
			this.A.push(value);
		}
	}
}
```
